struct dsu
{
    vector<int> sz, par;
    dsu(int n)
    {
        sz.assign(n + 1, 1);
        par.assign(n + 1, {});
        iota(par.begin(), par.end(), 0);
    }
    int find(int u)
    {
        if (par[u] == u)
            return u;
        return par[u] = find(par[u]);
    }
    void merge(int u, int v)
    {
        u = find(u), v = find(v);
        if (u == v)
        {
            return;
        }
        if (sz[u] < sz[v])
        {
            swap(u, v);
        }
        sz[u] += sz[v];
        par[v] = u;
    }
    bool iscon(int u,int v){
        u = find(u), v = find(v);
        return u == v;
    }
    int szco(int u)
    {
        u = find(u);
        return sz[u];
    }
};
