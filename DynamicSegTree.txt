struct DynamicSegTree {
    struct Node {
        int val = 0;
        Node* left = nullptr;
        Node* right = nullptr;
    };
 
    int neutral = 1; // if sm -> 0 ,if  mn -> 1e18 , mx -> -1e18 other with
    int size = 1e9;  // Set based on your use case range
    Node* root = new Node();
 
    int merge(int a, int b) {
        return a + b ;  // lc + rc 
    }
 
    void set(Node*& node, int idx, int val, int lx, int rx) {
        if (!node) node = new Node();
        if (lx == rx) {
            node->val = val; // change assign -> =  , ad -> += 
            return;
        }
 
        int mid = (lx + rx) >> 1;
        if (idx <= mid) set(node->left, idx, val, lx, mid);
        else set(node->right, idx, val, mid + 1, rx);
 
        int leftVal = node->left ? node->left->val : 0;
        int rightVal = node->right ? node->right->val : 0;
        node->val = merge(leftVal, rightVal);
    }
 
    void set(int idx, int val) {
        set(root, idx, val, 0, size);
    }
 
    int query(Node* node, int l, int r, int lx, int rx) {
        if (lx > r || rx < l) return neutral;
        if(!node) return 0;
        if (lx >= l && rx <= r) return node->val;
 
        int mid = (lx + rx) >> 1;
        int q1 = query(node->left, l, r, lx, mid);
        int q2 = query(node->right, l, r, mid + 1, rx);
        return merge(q1, q2);
    }
 
    int query(int l, int r) {
        return query(root, l, r, 0, size);
    }
};
